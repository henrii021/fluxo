<script>
(function () {
  // ===========================================================
  // AJUSTE OS PREÇOS AQUI (valores mensais em R$).
  // Steppers = preço por unidade. Toggles = preço fixo do serviço.
  // Os números abaixo são EXEMPLO, troque pelos seus reais.
  // ===========================================================
  var PRECOS = {
    posts:      35,   // por post no feed
    reels:      90,   // por reel
    stories:    40,   // por story/semana
    gestao:     150,  // gestão e publicação
    identidade: 120,  // identidade visual premium
    copy:       100,  // copy e legendas
    relatorio:  80,   // relatório mensal
    trafego:    250   // gestão de tráfego pago
  };
  var NOMES = {
    posts: 'Posts no feed', reels: 'Reels', stories: 'Stories/semana',
    gestao: 'Gestão e publicação', identidade: 'Identidade visual premium',
    copy: 'Copy e legendas', relatorio: 'Relatório mensal', trafego: 'Tráfego pago'
  };
  var WHATS = '5521975359273';

  var estado = { posts: 0, reels: 0, stories: 0, gestao: false, identidade: false, copy: false, relatorio: false, trafego: false };
  var root = document.getElementById('montar');
  if (!root) return;

  function fmt(n) { return 'R$ ' + n.toLocaleString('pt-BR'); }

  function render() {
    var lista = document.getElementById('fx-b-list');
    var linhas = [];
    var total = 0;
    ['posts','reels','stories'].forEach(function (k) {
      if (estado[k] > 0) {
        var v = estado[k] * PRECOS[k]; total += v;
        linhas.push('<div class="fx-b-line"><span>' + estado[k] + 'x ' + NOMES[k] + '</span><span>' + fmt(v) + '</span></div>');
      }
    });
    ['gestao','identidade','copy','relatorio','trafego'].forEach(function (k) {
      if (estado[k]) {
        total += PRECOS[k];
        linhas.push('<div class="fx-b-line"><span>' + NOMES[k] + '</span><span>' + fmt(PRECOS[k]) + '</span></div>');
      }
    });
    lista.innerHTML = linhas.length ? linhas.join('') : '<p class="fx-b-empty">Selecione os itens ao lado para montar o seu plano.</p>';
    document.getElementById('fx-b-total').textContent = fmt(total);
  }

  root.querySelectorAll('[data-type="step"]').forEach(function (item) {
    var key = item.getAttribute('data-key');
    var countEl = item.querySelector('[data-count]');
    item.querySelector('.fx-b-plus').addEventListener('click', function () {
      estado[key]++; countEl.textContent = estado[key];
      item.classList.toggle('fx-b-active', estado[key] > 0); render();
    });
    item.querySelector('.fx-b-minus').addEventListener('click', function () {
      if (estado[key] > 0) estado[key]--; countEl.textContent = estado[key];
      item.classList.toggle('fx-b-active', estado[key] > 0); render();
    });
  });

  root.querySelectorAll('[data-type="toggle"]').forEach(function (item) {
    var key = item.getAttribute('data-key');
    var input = item.querySelector('input');
    input.addEventListener('change', function () {
      estado[key] = input.checked;
      item.classList.toggle('fx-b-active', input.checked); render();
    });
  });

  document.getElementById('fx-b-cta').addEventListener('click', function () {
    var linhas = [], total = 0;
    ['posts','reels','stories'].forEach(function (k) { if (estado[k] > 0) { total += estado[k]*PRECOS[k]; linhas.push('• ' + estado[k] + 'x ' + NOMES[k]); } });
    ['gestao','identidade','copy','relatorio','trafego'].forEach(function (k) { if (estado[k]) { total += PRECOS[k]; linhas.push('• ' + NOMES[k]); } });
    if (!linhas.length) { alert('Selecione pelo menos um item para montar o seu pacote.'); return; }
    var texto = 'Olá! Montei meu pacote no site da Fluxo:\n\n' + linhas.join('\n') + '\n\nTotal estimado: ' + fmt(total) + '/mês (sem fidelidade).\n\nGostaria de fechar ou ajustar esse plano.';
    window.open('https://wa.me/' + WHATS + '?text=' + encodeURIComponent(texto), '_blank');
  });

  render();
})();
</script>
